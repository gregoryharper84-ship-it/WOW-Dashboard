def llp_mlb_winner_preflight(row):
    blockers = []
    # Gate 1: starter / lineup
    if row["starter_status"] not in ["CONFIRMED", "PROBABLE_STRONG"]:
        blockers.append("NO_STARTER_CONFIRMATION")
    if row["lineup_status"] not in ["CONFIRMED", "PROJECTED_ACCEPTABLE"]:
        blockers.append("NO_LINEUP_CONFIRMATION")
    # Gate 2: weather / event status
    if row["event_status"] in ["POSTPONED", "CANCELLED", "SUSPENDED"]:
        blockers.append("EVENT_RESET_REQUIRED_POSTPONEMENT")
    if row["weather_status"] in ["MATERIAL_RISK", "DELAY_RISK", "RAINOUT_RISK"]:
        blockers.append("WEATHER_RISK_CUT")
    # Gate 3: no-vig/model breakeven
    breakeven = 1 / row["kalshi_multiplier"]
    model_lb = row["calibrated_probability_lower_bound"]
    no_vig = row["sportsbook_no_vig_probability"]
    buffer = 0.02 if row["kalshi_multiplier"] < 1.60 else 0.015
    if no_vig < breakeven:
        blockers.append("NO_VIG_BELOW_BREAKEVEN")
    if model_lb < breakeven + buffer:
        blockers.append("MODEL_LOWER_BOUND_BELOW_BREAKEVEN")
    if blockers:
        row["preflight_status"] = "FAIL"
        row["can_upgrade"] = False
        row["blockers"] = blockers
        row["final_ceiling"] = "WINNER_WATCH_OR_REJECT"
    else:
        row["preflight_status"] = "PASS"
        row["can_upgrade"] = True
    return row
