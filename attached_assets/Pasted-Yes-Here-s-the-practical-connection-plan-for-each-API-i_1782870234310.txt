### Replit build instruction

Add a helper like:

```python
def nws_get(path_or_url: str) -> dict:
    import os
    import requests

    base = "https://api.weather.gov"
    url = path_or_url if path_or_url.startswith("http") else f"{base}{path_or_url}"

    headers = {
        "User-Agent": os.environ.get(
            "NWS_USER_AGENT",
            "WOW-v16-Kalshi-Weather contact@example.com"
        ),
        "Accept": "application/geo+json",
    }

    r = requests.get(url, headers=headers, timeout=15)
    r.raise_for_status()
    return r.json()
```

```python
CLI_ISSUEDBY = {
    "NYC": "NYC",
    "LA": "LAX",
    "MIA": "MIA",
    "CHI": "MDW",
    "AUS": "AUS",
}

def fetch_nws_cli(city: str) -> dict:
    import requests

    issuedby = CLI_ISSUEDBY[city.upper()]
    url = (
        "https://forecast.weather.gov/product.php"
        f"?site=NWS&issuedby={issuedby}&product=CLI&format=CI&version=1&glossary=0"
    )

    r = requests.get(url, timeout=15)
    r.raise_for_status()

    text = r.text
    return {
        "city": city.upper(),
        "issuedby": issuedby,
        "source": "nws_cli",
        "raw_text": text,
    }
```

```python
def ncei_cdo_get(endpoint: str, params: dict) -> dict:
    import os
    import requests

    token = os.environ.get("NOAA_CDO_TOKEN")
    if not token:
        return {
            "ok": False,
            "source_status": "DATA_UNOBTAINABLE",
            "reason": "NOAA_CDO_TOKEN missing",
        }

    base = "https://www.ncei.noaa.gov/cdo-web/api/v2"
    url = f"{base}/{endpoint.lstrip('/')}"

    headers = {"token": token}
    r = requests.get(url, headers=headers, params=params, timeout=20)
    r.raise_for_status()
    return r.json()
```

```python
def kalshi_get(path: str, params: dict | None = None) -> dict:
    import os
    import requests

    base = os.environ.get(
        "KALSHI_API_BASE",
        "https://external-api.kalshi.com/trade-api/v2"
    )
    url = f"{base}/{path.lstrip('/')}"

    r = requests.get(url, params=params or {}, timeout=15)
    r.raise_for_status()
    return r.json()


def fetch_kalshi_nhigh_markets(series_ticker: str) -> dict:
    return kalshi_get(
        "/markets",
        {
            "series_ticker": series_ticker,
            "status": "open",
            "limit": 100,
        },
    )
```

### Step B — fetch orderbook for ticker

```python
def fetch_kalshi_orderbook(ticker: str) -> dict:
    import datetime as dt

    data = kalshi_get(f"/markets/{ticker}/orderbook")
    now = dt.datetime.now(dt.timezone.utc).isoformat()

    orderbook = data.get("orderbook", {})
    yes_bids = orderbook.get("yes", []) or []
    no_bids = orderbook.get("no", []) or []

    best_yes_bid = max([level[0] for level in yes_bids], default=None)
    best_no_bid = max([level[0] for level in no_bids], default=None)

    # Kalshi uses cents. A NO bid at X implies a YES ask at 100 - X.
    best_yes_ask = (100 - best_no_bid) if best_no_bid is not None else None
    best_no_ask = (100 - best_yes_bid) if best_yes_bid is not None else None

    spread_cents = (
        best_yes_ask - best_yes_bid
        if best_yes_bid is not None and best_yes_ask is not None
        else None
    )

    return {
        "price_source": "kalshi_live_orderbook",
        "orderbook_source": "kalshi_public_market_data",
        "price_timestamp": now,
        "best_yes_bid": best_yes_bid / 100 if best_yes_bid is not None else None,
        "best_yes_ask": best_yes_ask / 100 if best_yes_ask is not None else None,
        "best_no_bid": best_no_bid / 100 if best_no_bid is not None else None,
        "best_no_ask": best_no_ask / 100 if best_no_ask is not None else None,
        "spread": spread_cents / 100 if spread_cents is not None else None,
        "live_orderbook_checked": True,
    }
```
