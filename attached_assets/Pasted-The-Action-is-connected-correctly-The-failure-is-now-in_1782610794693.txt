from flask import request, jsonify

@app.route("/wow/kalshi/scan", methods=["POST"])
def wow_kalshi_scan():
    payload = request.get_json(silent=True) or {}

    category = payload.get("category", "sports")
    sport = payload.get("sport")
    scan_date = payload.get("date")
    mode = payload.get("mode", "scan_only")
    min_adjusted_edge = payload.get("min_adjusted_edge", 0.04)
    max_markets = payload.get("max_markets", 50)

    # Safety lock
    DRY_RUN_ONLY = True
    ALLOW_LIVE_TRADING = False
    ALLOW_MARKET_ORDERS = False

    # Temporary safe stub until real Kalshi scanner is wired
    return jsonify({
        "markets_scanned": 0,
        "playable_limit_only": 0,
        "final_approved": 0,

            }
        ],
