## Express route example

If your legacy platform app uses Express as the GPT-safe proxy, add/repair this:

```js
app.get("/api/wow/kalshi/health", async (req, res) => {
  const base = {
    execution_rule: "DRY_RUN_ONLY_NO_LIVE_TRADING_NO_MARKET_ORDERS",
    can_execute: false
  };

  try {
    // Replace this with your actual Flask/internal health URL.
    const flaskUrl = process.env.FLASK_BASE_URL || "http://127.0.0.1:5000";
    const r = await fetch(`${flaskUrl}/api/wow/kalshi/health`, {
      method: "GET",
      headers: { "Accept": "application/json" }
    });

    let data = {};
    try {
      data = await r.json();
    } catch {
      return res.status(200).json({
        ok: false,
        signal: "FAILED",
        message: "Flask health route did not return JSON",
        upstream_status: r.status,
        ...base
      });
    }

    return res.status(200).json({
      ok: Boolean(data.ok),
      flask_reachable: true,
      kalshi_reachable: data.kalshi_reachable ?? null,
      signal: data.signal || "FAILED",
      inventory_count: data.inventory_count ?? null,
      mve_null_count: data.mve_null_count ?? null,
      mve_non_null_count: data.mve_non_null_count ?? null,
      deploy_version: data.deploy_version ?? null,
      message: data.message ?? null,
      ...base
    });
  } catch (err) {
    return res.status(200).json({
      ok: false,
      flask_reachable: false,
      kalshi_reachable: null,
      signal: "FAILED",
      message: "Kalshi health proxy failed",
      detail: String(err?.message || err),
      ...base
    });
  }
});
```
