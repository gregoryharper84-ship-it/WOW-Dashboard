```js
// API-Sports multi-sport (Baseball, Basketball, Hockey)
// Uses same APISPORTS_KEY env var as api-football

app.get('/api-sports/:sport/players', async (req, res) => {
  const { sport } = req.params;
  const { player, league, season } = req.query;
  const bases = {
    baseball: 'https://v1.baseball.api-sports.io',
    basketball: 'https://v1.basketball.api-sports.io',
    hockey: 'https://v1.hockey.api-sports.io',
  };
  const base = bases[sport];
  if (!base) return res.status(400).json({ error: 'Unknown sport' });
  const r = await fetch(
    `${base}/players?search=${encodeURIComponent(player)}&league=${league}&season=${season}`,
    { headers: { 'x-apisports-key': process.env.APISPORTS_KEY } }
  );
  res.json(await r.json());
});

app.get('/api-sports/:sport/stats', async (req, res) => {
  const { sport } = req.params;
  const { player_id, league, season } = req.query;
  const bases = {
    baseball: 'https://v1.baseball.api-sports.io',
    basketball: 'https://v1.basketball.api-sports.io',
    hockey: 'https://v1.hockey.api-sports.io',
  };
  const base = bases[sport];
  if (!base) return res.status(400).json({ error: 'Unknown sport' });
  const paths = {
    baseball: 'players/statistics',
    basketball: 'statistics/players',
    hockey: 'players/statistics',
  };
  const r = await fetch(
    `${base}/${paths[sport]}?id=${player_id}&league=${league}&season=${season}`,
    { headers: { 'x-apisports-key': process.env.APISPORTS_KEY } }
  );
  res.json(await r.json());
});
```
